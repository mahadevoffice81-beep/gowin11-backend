// ========================================== // 1. SAFE LOCALSTORAGE WRAPPER // ========================================== const Storage = { getItem: function(key) { try { return localStorage.getItem(key); } catch (e) { return window._memoryStore ? window._memoryStore[key] : null; } }, setItem: function(key, value) { try { localStorage.setItem(key, value); } catch (e) { if (!window._memoryStore) window._memoryStore = {}; window._memoryStore[key] = value; } } }; // ========================================== // 2. VARIABLE INITIALIZATIONS // ========================================== let isDemoUser = false; let currentUser = null; let registeredUsers = JSON.parse(Storage.getItem('gowin_registered_users') || '[]'); // ========================================== // 3. AUTHENTICATION & LOGIN SYSTEM // ========================================== function handleUserAuth(identifier, password, type = 'mobile') { identifier = identifier.trim(); password = password.trim(); let existingUser = registeredUsers.find(u => u.identifier === identifier && u.type === type); if (existingUser) { if (existingUser.password === password) { currentUser = existingUser; Storage.setItem('gowin_current_user', JSON.stringify(currentUser)); loadUserDashboard(currentUser); return { success: true, message: "Login successful!" }; } else { return { success: false, message: "Incorrect password! Please try again or use Forgot Password." }; } } else { let newUser = { uid: '#GW' + Math.floor(100000 + Math.random() * 900000), identifier: identifier, password: password, type: type, balance: 200.00, username: (type === 'mobile' ? 'MobileUser_' : 'EmailUser_') + Math.floor(1000 + Math.random() * 9000) }; registeredUsers.push(newUser); Storage.setItem('gowin_registered_users', JSON.stringify(registeredUsers)); currentUser = newUser; Storage.setItem('gowin_current_user', JSON.stringify(currentUser)); loadUserDashboard(currentUser); return { success: true, message: "Account created and logged in successfully!" }; } } function handleForgotPassword(identifier, type = 'mobile') { identifier = identifier.trim(); let existingUser = registeredUsers.find(u => u.identifier === identifier && u.type === type); if (existingUser) { return { success: true, message: `OTP sent to your ${type === 'mobile' ? 'Mobile Number' : 'Gmail'}. Your password is: ${existingUser.password}` }; } else { return { success: false, message: "This identifier is not registered with us!" }; } } function resetPassword(identifier, newPassword, type = 'mobile') { let userIndex = registeredUsers.findIndex(u => u.identifier === identifier && u.type === type); if (userIndex !== -1) { registeredUsers[userIndex].password = newPassword; Storage.setItem('gowin_registered_users', JSON.stringify(registeredUsers)); return { success: true, message: "Password updated successfully! Please login." }; } return { success: false, message: "User not found." }; } // ========================================== // 4. UI & DASHBOARD LOADER // ========================================== function loadUserDashboard(user) { isDemoUser = false; const usernameElement = document.querySelector('.username-display'); const uidElement = document.querySelector('.uid-display'); const balanceElement = document.querySelector('.balance-display'); if (usernameElement) usernameElement.innerText = user.username; if (uidElement) uidElement.innerText = `UID: ${user.uid}`; if (balanceElement) balanceElement.innerText = `${user.balance.toFixed(2)}`; } // Auto-login check on page load window.addEventListener('DOMContentLoaded', () => { try { const savedUser = Storage.getItem('gowin_current_user'); if (savedUser) { currentUser = JSON.parse(savedUser); loadUserDashboard(currentUser); } } catch (e) { console.log("No active session found."); } });
